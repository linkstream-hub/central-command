// NEW: Helper function to check trainee solo assignment
function checkTraineeWarning(assignedTechs: Tech[]): string | null {
  // assignedTechs = array of assigned Tech objects, each with rank: 'T' | 'C' | 'L' | 'L1' | 'L2'
  const hasTrainee = assignedTechs.some((t) => t.rank === 'T');
  const hasSupervisor = assignedTechs.some((t) => ['C', 'L', 'L1', 'L2'].includes(t.rank));
  
  if (hasTrainee && !hasSupervisor) {
    return 'Trainee must be paired with a Captain or Lieutenant';
  }
  return null;
}

      <AlertCircle className="h-4 w-4" />
      <span>{traineeWarning}</span>
    </div>
  );
}

WHAT NOT TO DO
Banned Patterns & Libraries
Do NOT hardcode hex colors in component files. Use Tailwind tokens only:
✅ text-red-500, bg-amber-500/20, border-white/10
❌ text-[#ff0000], style={{ color: '#ff0000' }}
Do NOT use alert(), window.confirm(), or native dialog APIs. Use toast notifications:
✅ toast.success("Job assigned"), toast.error("Assignment failed")
❌ alert("Job assigned")
Do NOT add new npm dependencies without explicit Claude Code approval. Flag any package you're unsure about.
Do NOT move or rename .gs, .js, or .html files at repo root. Clasp deployment will break.
Do NOT modify src/lib/dashboard-api.ts or src/lib/types.ts for this sprint unless a new API endpoint or type is explicitly required.
Do NOT touch dispatch office-only pages:
src/app/live/
src/app/schedule/
src/app/weekly-schedule/
src/components/dashboard/
Do NOT change authentication logic in auth.ts or push/subscribe/route.ts without coordinating with DashboardAPI.gs updates.
Do NOT remove Skeleton loaders from data-fetch states. Every async load needs a placeholder.
Do NOT use hardcoded animation durations. Use Tailwind transition utilities:
✅ transition-all duration-200, duration-300
❌ style={{ transition: 'all 0.2s ease' }}
Do NOT test glassmorphism on desktop only. Every UI change must be verified at 375px mobile width.
